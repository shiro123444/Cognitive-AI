     1	import axios from 'axios';
     2	
     3	const TOKEN_STORAGE_KEY = 'edufish.auth.token';
     4	const USER_STORAGE_KEY = 'edufish.auth.user';
     5	
     6	function readStoredToken() {
     7	  if (typeof localStorage === 'undefined') return '';
     8	  try {
     9	    return localStorage.getItem(TOKEN_STORAGE_KEY) || '';
    10	  } catch {
    11	    return '';
    12	  }
    13	}
    14	
    15	function extractMessage(envelopeError) {
    16	  if (!envelopeError) return 'Request failed';
    17	  if (typeof envelopeError === 'string') return envelopeError;
    18	  if (typeof envelopeError === 'object' && typeof envelopeError.message === 'string') {
    19	    return envelopeError.message;
    20	  }
    21	  return 'Request failed';
    22	}
    23	
    24	export function unwrapEnvelope(response) {
    25	  const payload = response?.data;
    26	
    27	  if (payload && typeof payload === 'object' && typeof payload.success === 'boolean') {
    28	    if (payload.success) {
    29	      return payload.data;
    30	    }
    31	
    32	    throw new Error(extractMessage(payload.error) || payload.message || 'Request failed');
    33	  }
    34	
    35	  return payload;
    36	}
    37	
    38	export function unwrapEnvelopeError(error) {
    39	  const payload = error?.response?.data;
    40	
    41	  if (payload && typeof payload === 'object' && typeof payload.success === 'boolean') {
    42	    const err = new Error(extractMessage(payload.error) || payload.message || 'Request failed');
    43	    err.status = error?.response?.status;
    44	    err.code = typeof payload.error === 'object' ? payload.error?.code : undefined;
    45	    throw err;
    46	  }
    47	
    48	  throw error;
    49	}
    50	
    51	const apiClient = axios.create({
    52	  baseURL: '',
    53	  timeout: 20000
    54	});
    55	
    56	apiClient.interceptors.request.use((config) => {
    57	  const token = readStoredToken();
    58	  if (token) {
    59	    config.headers = config.headers || {};
    60	    if (!config.headers.Authorization) {
    61	      config.headers.Authorization = `Bearer ${token}`;
    62	    }
    63	  }
    64	  return config;
    65	});
    66	
    67	apiClient.interceptors.response.use(unwrapEnvelope, (error) => {
    68	  const status = error?.response?.status;
    69	  if (status === 401 && typeof window !== 'undefined') {
    70	    try {
    71	      localStorage.removeItem(TOKEN_STORAGE_KEY);
    72	      localStorage.removeItem(USER_STORAGE_KEY);
    73	    } catch {
    74	      // localStorage may be unavailable in some environments
    75	    }
    76	    window.dispatchEvent(new CustomEvent('edufish:auth-expired'));
    77	  }
    78	  return unwrapEnvelopeError(error);
    79	});
    80	
    81	export { TOKEN_STORAGE_KEY, USER_STORAGE_KEY };
    82	export default apiClient;

     1	<script setup>
     2	import { computed, onBeforeUnmount, onMounted, ref } from 'vue';
     3	
     4	import { createRuntimeSession, listRuntimeEvents, startRuntimeRun } from '../api/runtime';
     5	import { buildRuntimeInspectorModel } from './runtimeInspectorState';
     6	
     7	const SESSION_KEY = 'edufish.runtime.session_id';
     8	const POLL_MS = 3000;
     9	
    10	const session = ref(null);
    11	const events = ref([]);
    12	const runs = ref([]);
    13	const error = ref('');
    14	const isBusy = ref(false);
    15	const isRunning = ref(false);
    16	let pollTimer = null;
    17	
    18	const model = computed(() => buildRuntimeInspectorModel({
    19	  session: session.value,
    20	  events: events.value,
    21	  runs: runs.value,
    22	  error: error.value
    23	}));
    24	
    25	function rememberSession(id) {
    26	  if (id && typeof localStorage !== 'undefined') {
    27	    try { localStorage.setItem(SESSION_KEY, id); } catch { /* storage may be unavailable */ }
    28	  }
    29	}
    30	
    31	async function ensureSession() {
    32	  let stored = '';
    33	  try { stored = typeof localStorage !== 'undefined' ? localStorage.getItem(SESSION_KEY) || '' : ''; } catch { /* noop */ }
    34	  if (stored) {
    35	    session.value = { session_id: stored };
    36	    return stored;
    37	  }
    38	  const created = await createRuntimeSession(['user:inspector']);
    39	  const id = created?.session_id || '';
    40	  rememberSession(id);
    41	  session.value = created;
    42	  return id;
    43	}
    44	
    45	async function refreshEvents() {
    46	  const id = session.value?.session_id;
    47	  if (!id) return;
    48	  const lastSeq = events.value.at(-1)?.session_seq ?? 0;
    49	  const res = await listRuntimeEvents(id, lastSeq);
    50	  const fresh = res?.events ?? [];
    51	  if (fresh.length) events.value = [...events.value, ...fresh];
    52	}
    53	
    54	async function bootstrap() {
    55	  isBusy.value = true;
    56	  error.value = '';
    57	  try {
    58	    await ensureSession();
    59	    await refreshEvents();
    60	    pollTimer = setInterval(refreshEvents, POLL_MS);
    61	  } catch (err) {
    62	    error.value = err?.message || 'Runtime 不可达：需启动 Node Runtime 服务，并经 nginx 反代 /runtime。';
    63	  } finally {
    64	    isBusy.value = false;
    65	  }
    66	}
    67	
    68	async function triggerRun() {
    69	  const id = session.value?.session_id;
    70	  if (!id || isRunning.value) return;
    71	  isRunning.value = true;
    72	  error.value = '';
    73	  try {
    74	    await startRuntimeRun({
    75	      session_id: id,
    76	      agent_id: 'tutor',
    77	      system_prompt: 'You are a runtime smoke-test agent.',
    78	      user_message: 'ping'
    79	    });
    80	    runs.value = [...runs.value, { at: Date.now() }];
    81	    await refreshEvents();
    82	  } catch (err) {
    83	    error.value = err?.message || 'Run 失败。';
    84	  } finally {
    85	    isRunning.value = false;
    86	  }
    87	}
    88	
    89	onMounted(bootstrap);
    90	onBeforeUnmount(() => { if (pollTimer) clearInterval(pollTimer); });
    91	</script>

     1	<script setup>
     2	import { Niivue } from '@niivue/niivue';
     3	import { nextTick, onBeforeUnmount, onMounted, ref, watch } from 'vue';
     4	
     5	const props = defineProps({
     6	  model: {
     7	    type: Object,
     8	    required: true
     9	  },
    10	  cameraResetToken: {
    11	    type: Number,
    12	    default: 0
    13	  }
    14	});
    15	
    16	const emit = defineEmits(['scene-ready', 'scene-error']);
    17	
    18	const canvas = ref(null);
    19	const status = ref('booting');
    20	
    21	let nv = null;
    22	let prevImagesKey = '';
    23	
    24	function applyCameraPreset() {
    25	  if (!nv || !props.model?.cameraPreset) return;
    26	  nv.setRenderAzimuthElevation(
    27	    props.model.cameraPreset.azimuth,
    28	    props.model.cameraPreset.elevation
    29	  );
    30	}
    31	
    32	function applyConnectome() {
    33	  // Drive the 3D brain region colors from band-power (connectome node Color).
    34	  if (!nv || !nv.loadConnectome || !props.model?.connectome) return;
    35	  try {
    36	    nv.loadConnectome(props.model.connectome);
    37	  } catch {
    38	    // connectome overlay is optional — never fail the scene on it
    39	  }
    40	}
    41	
    42	function imagesKey(images) {
    43	  return (images || []).map((img) => img.url || img).join('|');
    44	}
    45	
    46	async function mountScene() {
    47	  if (!canvas.value) return;
    48	
    49	  const newKey = imagesKey(props.model?.images);
    50	
    51	  // Same brain, new data (e.g. scrubber moved) — refresh connectome + camera only.
    52	  if (nv && prevImagesKey === newKey) {
    53	    applyCameraPreset();
    54	    applyConnectome();
    55	    return;
    56	  }
    57	
    58	  status.value = 'booting';
    59	
    60	  try {
    61	    nv?.cleanup?.();
    62	    nv = new Niivue({
    63	      backColor: [0, 0, 0, 0],
    64	      show3Dcrosshair: false,
    65	      isOrientCube: false,
    66	      crosshairWidth: 0
    67	    });
    68	
    69	    await nextTick();
    70	    await nv.attachToCanvas(canvas.value);
    71	    await nv.loadImages(props.model?.images || []);
    72	    if (nv.setSliceType && nv.sliceTypeRender != null) {
    73	      nv.setSliceType(nv.sliceTypeRender);
    74	    }
    75	    prevImagesKey = newKey;
    76	    applyCameraPreset();
    77	    applyConnectome();
    78	
    79	    status.value = 'ready';
    80	    emit('scene-ready');
    81	  } catch (error) {
    82	    status.value = 'error';
    83	    emit('scene-error', error?.message || 'niivue init failed');
    84	  }
    85	}
    86	
    87	// Expose the nv instance + connectome refresh so parents can drive runtime
    88	// updates (e.g. scrubber) without re-mounting the WebGL scene.
    89	defineExpose({
    90	  getNv: () => nv,
    91	  refreshConnectome: applyConnectome
    92	});
    93	
    94	watch(() => props.model.sceneRevision, mountScene);
    95	watch(() => props.cameraResetToken, applyCameraPreset);
    96	
    97	onMounted(mountScene);
    98	
    99	onBeforeUnmount(() => {
   100	  nv?.cleanup?.();
   101	});
   102	</script>

   336	    @staticmethod
   337	    def _tokenize(text):
   338	        """Extract search tokens supporting both ASCII and CJK text."""
   339	        import re
   340	
   341	        _STOPWORDS = {"a", "an", "and", "are", "as", "for", "how", "in", "is", "of", "or", "the", "to", "what",
   342	                      "的", "是", "在", "和", "了", "有", "我", "不", "人", "这", "他", "她", "它", "们", "那", "都",
   343	                      "也", "就", "要", "会", "可", "能", "对", "与", "及", "或", "但", "而", "且", "被", "把", "从",
   344	                      "让", "给", "向", "往", "朝", "沿", "按", "照", "以", "为", "因", "所", "者", "此", "其", "之"}
   345	        tokens = set()
   346	
   347	        # ASCII words (3+ chars)
   348	        for t in re.findall(r"[a-z0-9]{3,}", text.lower()):
   349	            if t not in _STOPWORDS:
   350	                tokens.add(t)
   351	
   352	        # CJK bigrams for better matching precision
   353	        cjk = re.findall(r"[一-鿿㐀-䶿]", text)
   354	        for i in range(len(cjk)):
   355	            char = cjk[i]
   356	            if char not in _STOPWORDS:
   357	                tokens.add(char)
   358	            if i < len(cjk) - 1:
   359	                tokens.add(cjk[i] + cjk[i + 1])
   360	
   361	        return tokens

   363	    @staticmethod
   364	    def _text_tokens(text):
   365	        """Extract tokens from a target text for matching against query tokens."""
   366	        import re
   367	
   368	        tokens = set()
   369	        for t in re.findall(r"[a-z0-9]{3,}", text.lower()):
   370	            tokens.add(t)
   371	        for ch in re.findall(r"[一-鿿㐀-䶿]", text):
   372	            tokens.add(ch)
   373	        return tokens

   375	    @staticmethod
   376	    def _graph_context(question, course_id, chapter_id, concept_id, user_id=""):
   377	        """Get relevant context from the knowledge graph (keyword match as supplement)."""
   378	        query_tokens = TutorService._tokenize(question)
   379	
   380	        if not query_tokens:
   381	            return ""
   382	
   383	        graph = CourseService.get_graph(
   384	            course_id=course_id,
   385	            owner_id=user_id or "",
   386	            include_personal=bool(user_id),
   387	        )
   388	        concepts_by_id = {node["id"]: node for node in graph["nodes"]}
   389	        parts = []
   390	        seen = set()
   391	
   392	        for edge in graph["edges"]:
   393	            source = concepts_by_id.get(edge["source"], {})
   394	            target = concepts_by_id.get(edge["target"], {})
   395	            if concept_id and concept_id not in {edge["source"], edge["target"]}:
   396	                continue
   397	            text = f"{source.get('label', '')} {source.get('definition', '')} {target.get('label', '')} {target.get('definition', '')} {edge.get('relationship', '')} {edge.get('evidence', '')}"
   398	            text_tokens = TutorService._text_tokens(text)
   399	            if len(query_tokens & text_tokens) >= min(2, len(query_tokens)):
   400	                key = f"{source.get('label', '?')} --[{edge.get('relationship', '')}]--> {target.get('label', '?')}"
   401	                if key not in seen:
   402	                    seen.add(key)
   403	                    parts.append(f"{key}: {edge.get('evidence', '')}")
   404	
   405	        for concept in graph["nodes"]:
   406	            if concept_id and concept["id"] != concept_id:
   407	                continue
   408	            text = f"{concept['label']} {concept.get('definition', '')}"
   409	            text_tokens = TutorService._text_tokens(text)
   410	            if len(query_tokens & text_tokens) >= min(2, len(query_tokens)):
   411	                key = f"概念: {concept['label']}"
   412	                if key not in seen:
   413	                    seen.add(key)
   414	                    parts.append(f"{key} — {concept.get('definition', '')}")
   415	
   416	        return "\n".join(parts[:5])

   418	    @staticmethod
   419	    def _keyword_answer(question, course_id=None, chapter_id=None, concept_id=None, user_id=""):
   420	        """Fallback: keyword matching (original behavior)."""
   421	        profile = TutorService.course_profile(course_id)
   422	        query_tokens = TutorService._tokenize(question)
   423	
   424	        def _matches(text):
   425	            if not query_tokens:
   426	                return False
   427	            return len(query_tokens & TutorService._text_tokens(text or "")) >= min(2, len(query_tokens))
   428	
   429	        def _snippet(*parts):
   430	            return " ".join(p for p in parts if p).strip()[:240]
   431	
   432	        graph = CourseService.get_graph(
   433	            course_id=course_id,
   434	            owner_id=user_id or "",
   435	            include_personal=bool(user_id),
   436	        )
   437	        concepts_by_id = {node["id"]: node for node in graph["nodes"]}
   438	        citations = []
   439	
   440	        for edge in graph["edges"]:
   441	            source = concepts_by_id.get(edge["source"], {})
   442	            target = concepts_by_id.get(edge["target"], {})
   443	            if concept_id and concept_id not in {edge["source"], edge["target"]}:
   444	                continue
   445	            text = f"{source.get('label', '')} {source.get('definition', '')} {target.get('label', '')} {target.get('definition', '')} {edge.get('relationship', '')} {edge.get('evidence', '')}"
   446	            if _matches(text):
   447	                citations.append({
   448	                    "type": "graph_edge",
   449	                    "id": edge["id"],
   450	                    "title": f"{source.get('label', edge['source'])} {edge['relationship']} {target.get('label', edge['target'])}",
   451	                    "snippet": _snippet(edge.get("evidence", "")),
   452	                })
   453	
   454	        for concept in graph["nodes"]:
   455	            if concept_id and concept["id"] != concept_id:
   456	                continue
   457	            if _matches(f"{concept['label']} {concept['definition']}"):
   458	                citations.append({
   459	                    "type": "concept",
   460	                    "id": concept["id"],
   461	                    "title": concept["label"],
   462	                    "snippet": _snippet(concept.get("definition", "")),
   463	                })

===== backend/app/agents/registry.py =====

     1	"""Tool registry for agent tool calling.
     2	
     3	Tools self-register at import time. Each tool has:
     4	- name: unique identifier
     5	- description: what it does (shown to LLM)
     6	- parameters: JSONSchema for arguments
     7	- handler: callable that executes the tool
     8	
     9	The registry is global and immutable after import.
    10	"""
    11	
    12	from __future__ import annotations
    13	
    14	from dataclasses import dataclass, field
    15	from typing import Callable
    16	
    17	
    18	@dataclass
    19	class Tool:
    20	    """A tool that an agent can call."""
    21	
    22	    name: str
    23	    description: str
    24	    parameters: dict  # JSONSchema for tool arguments
    25	    handler: Callable[..., dict]
    26	    requires_app_context: bool = True
    27	
    28	    def to_openai_schema(self) -> dict:
    29	        """Convert to OpenAI function/tool schema."""
    30	        return {
    31	            "type": "function",
    32	            "function": {
    33	                "name": self.name,
    34	                "description": self.description,
    35	                "parameters": self.parameters,
    36	            },
    37	        }
    38	
    39	
    40	@dataclass
    41	class ToolRegistry:
    42	    """Global registry of available tools."""
    43	
    44	    _tools: dict[str, Tool] = field(default_factory=dict)
    45	
    46	    def register(self, tool: Tool) -> None:
    47	        if tool.name in self._tools:
    48	            raise ValueError(f"Tool already registered: {tool.name}")
    49	        self._tools[tool.name] = tool
    50	
    51	    def get(self, name: str) -> Tool | None:
    52	        return self._tools.get(name)
    53	
    54	    def list_names(self) -> list[str]:
    55	        return sorted(self._tools.keys())
    56	
    57	    def schemas_for(self, names: list[str]) -> list[dict]:
    58	        """Get OpenAI schemas for a subset of tools."""
    59	        return [self._tools[n].to_openai_schema() for n in names if n in self._tools]
    60	
    61	    def all_schemas(self) -> list[dict]:
    62	        return [t.to_openai_schema() for t in self._tools.values()]
    63	
    64	
    65	# Global registry instance
    66	registry = ToolRegistry()
    67	
    68	
    69	def register_tool(name: str, description: str, parameters: dict, requires_app_context: bool = True):
    70	    """Decorator to register a function as a tool."""
    71	
    72	    def decorator(func: Callable[..., dict]) -> Callable[..., dict]:
    73	        tool = Tool(
    74	            name=name,
    75	            description=description,
    76	            parameters=parameters,
    77	            handler=func,
    78	            requires_app_context=requires_app_context,
    79	        )
    80	        registry.register(tool)
    81	        return func
    82	
    83	    return decorator

     1	import { randomUUID } from 'node:crypto';

    12	interface EventRow {
    13	  event_id: string;
    14	  session_id: string;
    15	  run_id: string;
    16	  session_seq: number | string;
    17	  type: string;
    18	  payload_json: Record<string, unknown>;
    19	  timestamp: string | Date;
    20	}
    21	
    22	/**
    23	 * Append-only event store.
    24	 *
    25	 * session_seq is allocated with SELECT max+1. Parallel child runs (P1.5 fan-out)
    26	 * share a session, so appends are serialized via an in-process mutex to avoid
    27	 * primary-key collisions. A multi-instance deployment would need a DB sequence
    28	 * or advisory lock instead.
    29	 */
    30	export class EventStore {
    31	  /** Chains appends so concurrent writers cannot race on session_seq. */
    32	  private appendTail: Promise<unknown> = Promise.resolve();
    33	
    34	  constructor(private readonly db: RuntimeDb) {}
    35	
    36	  async append(input: AppendEventInput) {
    37	    // Serialize: wait for previous append, hold the chain until we finish.
    38	    let release!: () => void;
    39	    const prev = this.appendTail;
    40	    this.appendTail = new Promise<void>((resolve) => {
    41	      release = resolve;
    42	    });
    43	    await prev;
    44	
    45	    try {
    46	      return await this.appendUnlocked(input);
    47	    } finally {
    48	      release();
    49	    }
    50	  }
    51	
    52	  private async appendUnlocked(input: AppendEventInput) {
    53	    const next = await this.db.query<{ next_seq: number | string }>(
    54	      'SELECT COALESCE(MAX(session_seq), 0) + 1 AS next_seq FROM events WHERE session_id = $1',
    55	      [input.session_id]
    56	    );
    57	
    58	    const event = {
    59	      event_id: randomUUID(),
    60	      session_id: input.session_id,
    61	      run_id: input.run_id,
    62	      session_seq: Number(next.rows[0].next_seq),
    63	      type: input.type,
    64	      payload: input.payload,
    65	      timestamp: new Date().toISOString()
    66	    };
    67	
    68	    await this.db.query(
    69	      `
    70	        INSERT INTO events (session_id, session_seq, event_id, run_id, type, payload_json, timestamp)
    71	        VALUES ($1, $2, $3, $4, $5, $6::jsonb, $7)
    72	      `,
    73	      [
    74	        event.session_id,
    75	        event.session_seq,
    76	        event.event_id,
    77	        event.run_id,
    78	        event.type,
    79	        JSON.stringify(event.payload),
    80	        event.timestamp
    81	      ]
    82	    );
    83	
    84	    return event;
    85	  }
    86	
    87	  async listSince(sessionId: string, lastSeenSeq: number) {
    88	    const result = await this.db.query<EventRow>(
    89	      `
    90	        SELECT event_id, session_id, run_id, session_seq, type, payload_json, timestamp
    91	        FROM events
    92	        WHERE session_id = $1 AND session_seq > $2
    93	        ORDER BY session_seq ASC
    94	      `,
    95	      [sessionId, lastSeenSeq]
    96	    );
    97	
    98	    return result.rows.map((row) => ({
    99	      event_id: row.event_id,
   100	      session_id: row.session_id,
   101	      run_id: row.run_id,
   102	      session_seq: Number(row.session_seq),
   103	      type: row.type,
   104	      payload: row.payload_json,
   105	      timestamp: new Date(row.timestamp).toISOString()
   106	    }));
   107	  }
   108	}

   187	// --- Implementation ---
   188	
   189	export class AgentLoop {
   190	  private readonly provider: LlmProvider;
   191	  private readonly capabilities: CapabilityClient;
   192	  private readonly eventStore: EventStore;
   193	  private readonly eventBus: EventBus<AgentLoopEvent>;
   194	  private readonly startChildRun?: StartChildRunFn;
   195	  private readonly maxDepth: number;
   196	
   197	  constructor(deps: AgentLoopDeps) {
   198	    this.provider = deps.provider;
   199	    this.capabilities = deps.capabilities;
   200	    this.eventStore = deps.eventStore;
   201	    this.eventBus = deps.eventBus;
   202	    this.startChildRun = deps.startChildRun;
   203	    this.maxDepth = deps.maxDepth ?? 3;
   204	  }
   205	
   206	  /**
   207	   * Execute a full agent run. Drives the FSM from 'queued' through to terminal state.
   208	   */
   209	  async execute(config: AgentLoopConfig): Promise<AgentLoopResult> {
   210	    const { sessionId, runId, signal, maxTurns = 20 } = config;
   211	    const depth = config.depth ?? 0;
   212	
   213	    // Keep as RunState (not a literal) so TS control-flow stays open across transitions.
   214	    let state: RunState = 'queued' as RunState;
   215	    let lastSummary = '';
   216	
   217	    const allowlist = resolveToolAllowlist(config.agentId, config.toolAllowlist);
   218	    // For enforcement on invoke: concrete set when restricted, null when unrestricted
   219	    const allowedToolSet: Set<string> | null =
   220	      allowlist === undefined || allowlist === '*' ? null : new Set(allowlist);
   221	
   222	    const mayDelegate =
   223	      Boolean(this.startChildRun) && depth + 1 < this.maxDepth && agentMayDelegate(config.agentId);
   224	
   225	    const transition = async (action: RunAction): Promise<RunState> => {
   226	      const from = state;
   227	      const to = nextRunState(from, action);
   228	      state = to;
   229	      const event: AgentLoopEvent = { type: 'run.state_changed', run_id: runId, from, to, action };
   230	      this.eventBus.emit(event);
   231	      await this.eventStore.append({
   232	        session_id: sessionId,
   233	        run_id: runId,
   234	        type: 'run.state_changed',
   235	        payload: {
   236	          from,
   237	          to,
   238	          action,
   239	          parent_run_id: config.parentRunId ?? null,
   240	          depth,
   241	          agent_id: config.agentId,
   242	        },
   243	      });
   244	      return to;
   245	    };
   246	
   247	    const end = (reason: 'completed' | 'failed' | 'cancelled' | 'max_turns'): AgentLoopResult => {
   248	      this.eventBus.emit({ type: 'agent.end', run_id: runId, reason });
   249	      return { state, summary: lastSummary, runId };
   250	    };
   251	
   252	    // Start the run
   253	    state = await transition('start');
   254	
   255	    // Discover available tools, then scope by agent catalog / grants
   256	    let toolDefs: LlmToolDef[];
   257	    try {
   258	      const capabilities = await this.capabilities.discover();
   259	      const all = capabilities.map((c: CapabilityDescriptor) => ({
   260	        name: c.capability_id,
   261	        description: c.description,
   262	        parameters: c.input_schema,
   263	      }));
   264	      toolDefs = filterToolsByAllowlist(all, allowlist);
   265	    } catch {
   266	      state = await transition('fail');
   267	      return end('failed');
   268	    }
   269	
   270	    if (mayDelegate) {
   271	      toolDefs = [...toolDefs, DELEGATE_TOOL_DEF];
   272	    }
   273	
   274	    // Build initial message history
   275	    const messages: LlmMessage[] = [
   276	      { role: 'system', content: config.systemPrompt },
   277	      { role: 'user', content: config.userMessage },
   278	    ];
   279	
   280	    let turn = 0;
   281	
   282	    // Main loop
   283	    while (turn < maxTurns) {
   284	      if (signal?.aborted) {
   285	        state = await transition('cancel');
   286	        return end('cancelled');
   287	      }
   288	
   289	      turn++;
   290	      this.eventBus.emit({ type: 'turn.start', run_id: runId, turn });
   291	
   292	      // LLM call
   293	      const assistantMessage = await this.provider.complete(messages, toolDefs, signal);
   294	      messages.push(assistantMessage);
   295	      if (assistantMessage.content) {
   296	        lastSummary = assistantMessage.content;
   297	      }
   298	
   299	      await this.eventStore.append({
   300	        session_id: sessionId,
   301	        run_id: runId,
   302	        type: 'llm.response',
   303	        payload: {
   304	          role: assistantMessage.role,
   305	          content: assistantMessage.content,
   306	          tool_calls: assistantMessage.tool_calls ?? null,
   307	        },
   308	      });
   309	      this.eventBus.emit({ type: 'llm.response', run_id: runId, message: assistantMessage });
   310	
   311	      const toolCalls = assistantMessage.tool_calls;
   312	
   313	      if (!toolCalls || toolCalls.length === 0) {
   314	        this.eventBus.emit({ type: 'turn.end', run_id: runId, turn });
   315	        state = await transition('complete');
   316	        return end('completed');
   317	      }
   318	
   319	      // Execute tool calls (capability tools and/or runtime.delegate)
   320	      for (const toolCall of toolCalls) {
   321	        if (signal?.aborted) break;
   322	
   323	        const isDelegate = toolCall.name === DELEGATE_TOOL_NAME;
   324	
   325	        // Switch FSM lane when tool kind changes mid-batch
   326	        if (isDelegate) {
   327	          if (state === 'waiting_tool') state = await transition('tool_complete');
   328	          if (state === 'running') state = await transition('delegate');
   329	        } else {
   330	          if (state === 'waiting_child') state = await transition('child_complete');
   331	          if (state === 'running') state = await transition('wait_tool');
   332	        }
   333	
   334	        this.eventBus.emit({
   335	          type: 'tool.start',
   336	          run_id: runId,
   337	          tool_call_id: toolCall.id,
   338	          name: toolCall.name,
   339	          arguments: toolCall.arguments,
   340	        });
   341	
   342	        await this.eventStore.append({
   343	          session_id: sessionId,
   344	          run_id: runId,
   345	          type: 'tool.start',
   346	          payload: { tool_call_id: toolCall.id, name: toolCall.name, arguments: toolCall.arguments },
   347	        });
   348	
   349	        let result: Record<string, unknown>;
   350	        let status: 'completed' | 'failed';
   351	
   352	        try {
   353	          if (isDelegate) {
   354	            if (!mayDelegate) {
   355	              result = { error: `agent ${config.agentId} is not allowed to delegate` };
   356	              status = 'failed';
   357	            } else {
   358	              const del = await this.handleDelegate(config, toolCall.arguments, depth);
   359	              result = del.result;
   360	              status = del.status;
   361	            }
   362	          } else if (allowedToolSet && !allowedToolSet.has(toolCall.name)) {
   363	            result = {
   364	              error: `tool ${toolCall.name} is not in the allowlist for agent ${config.agentId}`,
   365	            };
   366	            status = 'failed';
   367	          } else {
   368	            const capResult = await this.capabilities.invoke(toolCall.name, toolCall.arguments);
   369	            result = capResult.result;
   370	            status = capResult.status;
   371	          }
   372	        } catch (err) {
   373	          result = { error: err instanceof Error ? err.message : 'unknown error' };
   374	          status = 'failed';
   375	        }
   376	
   377	        this.eventBus.emit({
   378	          type: 'tool.end',
   379	          run_id: runId,
   380	          tool_call_id: toolCall.id,
   381	          name: toolCall.name,
   382	          status,
   383	          result,
   384	        });
   385	
   386	        await this.eventStore.append({
   387	          session_id: sessionId,
   388	          run_id: runId,
   389	          type: 'tool.end',
   390	          payload: { tool_call_id: toolCall.id, name: toolCall.name, status, result },
   391	        });
   392	
   393	        messages.push({
   394	          role: 'tool_result',
   395	          content: JSON.stringify(result),
   396	          tool_call_id: toolCall.id,
   397	        });
   398	      }
   399	
   400	      if (signal?.aborted) {
   401	        state = await transition('cancel');
   402	        return end('cancelled');
   403	      }
   404	
   405	      // Return to running from whichever waiting state we are in
   406	      if (state === 'waiting_child') {
   407	        state = await transition('child_complete');
   408	      } else if (state === 'waiting_tool') {
   409	        state = await transition('tool_complete');
   410	      }
   411	
   412	      this.eventBus.emit({ type: 'turn.end', run_id: runId, turn });
   413	    }
   414	
   415	    // Max turns reached
   416	    state = await transition('complete');
   417	    return end('max_turns');
   418	  }
   419	
   420	  /**
   421	   * Built-in runtime.delegate: plan → fan-out child runs → merge.
   422	   * Does NOT go through the Python capability bridge.
   423	   */
   424	  private async handleDelegate(
   425	    config: AgentLoopConfig,
   426	    args: Record<string, unknown>,
   427	    parentDepth: number,
   428	  ): Promise<{ status: 'completed' | 'failed'; result: Record<string, unknown> }> {
   429	    if (!this.startChildRun) {
   430	      return {
   431	        status: 'failed',
   432	        result: { error: 'runtime.delegate is not enabled on this runtime' },
   433	      };
   434	    }
   435	
   436	    const rawTasks = args.tasks;
   437	    if (!Array.isArray(rawTasks) || rawTasks.length === 0) {
   438	      return {
   439	        status: 'failed',
   440	        result: { error: 'runtime.delegate requires a non-empty tasks array' },
   441	      };
   442	    }
   443	
   444	    const tasks: DelegateTaskArg[] = rawTasks.map((t) => {
   445	      const item = t as Record<string, unknown>;
   446	      return {
   447	        to_agent_id: String(item.to_agent_id ?? item.toAgentId ?? ''),
   448	        goal: String(item.goal ?? ''),
   449	        system_prompt: item.system_prompt != null ? String(item.system_prompt) : undefined,
   450	        grant_ids: Array.isArray(item.grant_ids)
   451	          ? item.grant_ids.map(String)
   452	          : Array.isArray(item.grantIds)
   453	            ? item.grantIds.map(String)
   454	            : [],
   455	        constraints: Array.isArray(item.constraints) ? item.constraints.map(String) : [],
   456	      };
   457	    });
   458	
   459	    for (const t of tasks) {
   460	      if (!t.to_agent_id || !t.goal) {
   461	        return {
   462	          status: 'failed',
   463	          result: { error: 'each task requires to_agent_id and goal' },
   464	        };
   465	      }
   466	      if (!canDelegateTo(config.agentId, t.to_agent_id)) {
   467	        return {
   468	          status: 'failed',
   469	          result: {
   470	            error: `agent ${config.agentId} is not allowed to delegate to ${t.to_agent_id}`,
   471	          },
   472	        };
   473	      }
   474	    }
   475	
   476	    const plans = planDelegations({
   477	      fromRunId: config.runId,
   478	      tasks: tasks.map((t) => ({
   479	        toAgentId: t.to_agent_id,
   480	        goal: t.goal,
   481	        grantIds: t.grant_ids ?? [],
   482	        constraints: t.constraints,
   483	      })),
   484	    });
   485	
   486	    await this.eventStore.append({
   487	      session_id: config.sessionId,
   488	      run_id: config.runId,
   489	      type: 'delegation.planned',
   490	      payload: {
   491	        parent_run_id: config.runId,
   492	        count: plans.length,
   493	        plans,
   494	      },
   495	    });
   496	
   497	    // Fan-out: run all children in parallel
   498	    const childResults = await Promise.all(
   499	      tasks.map(async (task, i) => {
   500	        const plan = plans[i];
   501	
   502	        this.eventBus.emit({
   503	          type: 'delegation.start',
   504	          run_id: config.runId,
   505	          delegation_id: plan.delegation_id,
   506	          to_agent_id: task.to_agent_id,
   507	          goal: task.goal,
   508	        });
   509	
   510	        await this.eventStore.append({
   511	          session_id: config.sessionId,
   512	          run_id: config.runId,
   513	          type: 'delegation.start',
   514	          payload: {
   515	            delegation_id: plan.delegation_id,
   516	            to_agent_id: task.to_agent_id,
   517	            goal: task.goal,
   518	            context_grants: plan.context_grants,
   519	          },
   520	        });
   521	
   522	        // Grants can further restrict child tools (intersection with catalog)
   523	        const grantTools = toolAllowlistFromGrants(task.grant_ids);
   524	        const childCatalog = resolveToolAllowlist(task.to_agent_id);
   525	        let childAllow: string[] | null = null;
   526	        if (grantTools && childCatalog && childCatalog !== '*') {
   527	          const catalogSet = new Set(childCatalog);
   528	          childAllow = grantTools.filter((t) => catalogSet.has(t));
   529	        } else if (grantTools) {
   530	          childAllow = grantTools;
   531	        }
   532	
   533	        const child = await this.startChildRun!({
   534	          sessionId: config.sessionId,
   535	          parentRunId: config.runId,
   536	          agentId: task.to_agent_id,
   537	          systemPrompt: task.system_prompt ?? defaultSystemPrompt(task.to_agent_id),
   538	          userMessage: task.goal,
   539	          grantIds: task.grant_ids ?? [],
   540	          toolAllowlist: childAllow,
   541	          depth: parentDepth + 1,
   542	          signal: config.signal,
   543	        });
   544	
   545	        const childStatus: 'completed' | 'failed' =
   546	          child.finalState === 'completed' ? 'completed' : 'failed';
   547	
   548	        this.eventBus.emit({
   549	          type: 'delegation.end',
   550	          run_id: config.runId,
   551	          delegation_id: plan.delegation_id,
   552	          child_run_id: child.runId,
   553	          status: childStatus,
   554	        });
   555	
   556	        await this.eventStore.append({
   557	          session_id: config.sessionId,
   558	          run_id: config.runId,
   559	          type: 'delegation.end',
   560	          payload: {
   561	            delegation_id: plan.delegation_id,
   562	            child_run_id: child.runId,
   563	            status: childStatus,
   564	            summary: child.summary,
   565	            artifact_ref: child.artifactRef,
   566	          },
   567	        });
   568	
   569	        return {
   570	          run_id: child.runId,
   571	          artifact_ref: child.artifactRef,
   572	          status: childStatus,
   573	          summary: child.summary,
   574	          agent_id: task.to_agent_id,
   575	          delegation_id: plan.delegation_id,
   576	        };
   577	      }),
   578	    );
   579	
   580	    const mergeInputs: ChildResultInput[] = childResults.map((c) => ({
   581	      run_id: c.run_id,
   582	      artifact_ref: c.artifact_ref,
   583	      status: c.status,
   584	    }));
   585	    const merged = mergeChildResults(mergeInputs);
   586	
   587	    const status: 'completed' | 'failed' =
   588	      merged.failed === merged.child_count && merged.child_count > 0 ? 'failed' : 'completed';
   589	
   590	    return {
   591	      status,
   592	      result: {
   593	        ...merged,
   594	        children: childResults,
   595	      },
   596	    };
   597	  }
   598	}

    98	 * - Known agent with non-empty allowed_targets → yes
    99	 * - Unknown agent → yes (open default for custom agents)

   102	  const def = getAgentDef(agentId);
   103	  if (!def) return true;

   107	/**
   108	 * Resolve the effective tool allowlist for an agent run.
   109	 *
   110	 * Precedence:
   111	 * 1. Explicit toolAllowlist from the caller (e.g. parent-granted tools)
   112	 * 2. Agent catalog tools for agentId
   113	 * 3. undefined → no restriction (all discovered capabilities)
   114	 */
   115	export function resolveToolAllowlist(
   116	  agentId: string,
   117	  explicit?: string[] | null,
   118	): string[] | '*' | undefined {
   119	  if (explicit && explicit.length > 0) {
   120	    return explicit;
   121	  }
   122	  const def = getAgentDef(agentId);
   123	  if (!def) return undefined;
   124	  return def.tools;
   125	}

   127	/**
   128	 * Filter capability tool defs by allowlist.
   129	 */
   130	export function filterToolsByAllowlist<T extends { name: string }>(
   131	  tools: T[],
   132	  allowlist: string[] | '*' | undefined,
   133	): T[] {
   134	  if (allowlist === undefined || allowlist === '*') {
   135	    return tools;
   136	  }
   137	  const allowed = new Set(allowlist);
   138	  return tools.filter((t) => allowed.has(t.name));
   139	}
   140	
   141	/**
   142	 * Map grant_ids to a tool allowlist when grants encode tool scopes.
   143	 *
   144	 * Supported forms:
   145	 * - `tool:search_materials` → search_materials
   146	 * - bare capability id (`search_*`, `get_*`, `list_*`, `collect_*`, `trigger_*`, `check_*`, `runtime.*`)
   147	 * - otherwise ignored (opaque context grant refs)
   148	 */
   149	export function toolAllowlistFromGrants(grantIds: string[] | undefined): string[] | undefined {
   150	  if (!grantIds || grantIds.length === 0) return undefined;
   151	
   152	  const tools: string[] = [];
   153	  for (const g of grantIds) {
   154	    if (g.startsWith('tool:')) {
   155	      tools.push(g.slice('tool:'.length));
   156	      continue;
   157	    }
   158	    if (
   159	      g.startsWith('search_') ||
   160	      g.startsWith('get_') ||
   161	      g.startsWith('list_') ||
   162	      g.startsWith('collect_') ||
   163	      g.startsWith('trigger_') ||
   164	      g.startsWith('check_') ||
   165	      g.startsWith('runtime.')
   166	    ) {
   167	      tools.push(g);
   168	    }
   169	  }
   170	  return tools.length > 0 ? tools : undefined;
   171	}

   173	/** Validate that a supervisor may delegate to the given child agent. */
   174	export function canDelegateTo(fromAgentId: string, toAgentId: string): boolean {
   175	  const from = getAgentDef(fromAgentId);
   176	  if (!from) {
   177	    // Unknown parent: allow (open by default for custom agents)
   178	    return true;
   179	  }
   180	  if (from.allowed_targets.length === 0) return false;
   181	  if (from.allowed_targets.includes('*')) return true;
   182	  return from.allowed_targets.includes(toAgentId);
   183	}

   187	}

     3	export interface OpenAIProviderOptions {
     4	  baseUrl: string;
     5	  apiKey: string;
     6	  model: string;
     7	  /** Optional fetch override (for tests). Defaults to global fetch. */
     8	  fetchImpl?: typeof fetch;
     9	}
    10	
    11	interface OpenAIToolCall {
    12	  id: string;
    13	  type: 'function';
    14	  function: { name: string; arguments: string };
    15	}
    16	
    17	interface OpenAIChatMessage {
    18	  role: 'system' | 'user' | 'assistant' | 'tool';
    19	  content: string;
    20	  tool_call_id?: string;
    21	  tool_calls?: OpenAIToolCall[];
    22	}
    23	
    24	interface OpenAIChatResponse {
    25	  choices?: Array<{
    26	    message?: { content?: string | null; tool_calls?: OpenAIToolCall[] };
    27	  }>;
    28	  error?: { message?: string };
    29	}
    30	
    31	/**
    32	 * OpenAI-compatible providers (DeepSeek, some gateways) require tool names to
    33	 * match `^[a-zA-Z0-9_-]+$`. Our catalog uses dotted ids (`runtime.echo`,
    34	 * `runtime.delegate`). Encode dots for the wire format and restore them on
    35	 * the response so the agent loop keeps stable internal names.
    36	 */
    37	export function toWireToolName(name: string): string {
    38	  return name.replace(/\./g, '__');
    39	}
    40	
    41	export function fromWireToolName(wire: string, known?: ReadonlyMap<string, string>): string {
    42	  if (known?.has(wire)) return known.get(wire)!;
    43	  // Fallback for models that echo our encoding without a map hit.
    44	  return wire.includes('__') ? wire.replace(/__/g, '.') : wire;
    45	}
    46	
    47	function buildWireNameMap(tools: LlmToolDef[], messages: LlmMessage[]): Map<string, string> {
    48	  const wireToOriginal = new Map<string, string>();
    49	  const register = (name: string) => {
    50	    const wire = toWireToolName(name);
    51	    wireToOriginal.set(wire, name);
    52	  };
    53	  for (const t of tools) register(t.name);
    54	  for (const msg of messages) {
    55	    if (msg.role === 'assistant' && msg.tool_calls) {
    56	      for (const tc of msg.tool_calls) register(tc.name);
    57	    }
    58	  }
    59	  return wireToOriginal;
    60	}

    62	/**
    63	 * OpenAI-compatible LLM provider. Works with OpenAI, NVIDIA NIM, Xiaomi MiMo,
    64	 * Ollama /v1, DeepSeek, etc. Uses native fetch — no SDK dependency.
    65	 *
    66	 * Adapts the runtime's internal LlmMessage shape (role 'tool_result', flat
    67	 * tool_calls) to the OpenAI wire format (role 'tool', nested function calls).
    68	 * Per the LlmProvider contract, model/request errors are encoded into the
    69	 * returned message rather than thrown, so the agent loop never crashes on a
    70	 * provider failure.
    71	 */
    72	export class OpenAICompatibleProvider implements LlmProvider {
    73	  private readonly baseUrl: string;
    74	  private readonly apiKey: string;
    75	  private readonly model: string;
    76	  private readonly fetchImpl: typeof fetch;
    77	
    78	  constructor(opts: OpenAIProviderOptions) {
    79	    this.baseUrl = opts.baseUrl.replace(/\/$/, '');
    80	    this.apiKey = opts.apiKey;
    81	    this.model = opts.model;
    82	    this.fetchImpl = opts.fetchImpl ?? globalThis.fetch;
    83	  }
    84	
    85	  async complete(messages: LlmMessage[], tools: LlmToolDef[], signal?: AbortSignal): Promise<LlmMessage> {
    86	    const wireMap = buildWireNameMap(tools, messages);
    87	    const body: Record<string, unknown> = {
    88	      model: this.model,
    89	      messages: messages.map((m) => toOpenAIMessage(m, wireMap)),
    90	    };
    91	    if (tools.length > 0) {
    92	      body.tools = tools.map((t) => toOpenAITool(t, wireMap));
    93	      body.tool_choice = 'auto';
    94	    }
    95	
    96	    let data: OpenAIChatResponse;
    97	    try {
    98	      const res = await this.fetchImpl(`${this.baseUrl}/chat/completions`, {
    99	        method: 'POST',
   100	        headers: {
   101	          'Content-Type': 'application/json',
   102	          Authorization: `Bearer ${this.apiKey}`,
   103	        },
   104	        body: JSON.stringify(body),
   105	        signal,
   106	      });
   107	      data = (await res.json()) as OpenAIChatResponse;
   108	      if (!res.ok) {
   109	        return errorMessage(data?.error?.message || `LLM HTTP ${res.status}`);
   110	      }
   111	    } catch (err) {
   112	      const text = signal?.aborted
   113	        ? 'request aborted'
   114	        : err instanceof Error
   115	          ? err.message
   116	          : 'LLM request failed';
   117	      return errorMessage(text);
   118	    }
   119	
   120	    const choice = data.choices?.[0]?.message;
   121	    if (!choice) {
   122	      return errorMessage('LLM returned no choices');
   123	    }
   124	
   125	    const toolCalls = Array.isArray(choice.tool_calls)
   126	      ? choice.tool_calls.map((raw) => parseToolCall(raw, wireMap))
   127	      : undefined;
   128	
   129	    return {
   130	      role: 'assistant',
   131	      content: typeof choice.content === 'string' ? choice.content : '',
   132	      tool_calls: toolCalls && toolCalls.length > 0 ? toolCalls : undefined,
   133	    };
   134	  }
   135	}
   136	
   137	function toOpenAIMessage(msg: LlmMessage, wireMap: ReadonlyMap<string, string>): OpenAIChatMessage {
   138	  if (msg.role === 'tool_result') {
   139	    return { role: 'tool', content: msg.content, tool_call_id: msg.tool_call_id };
   140	  }
   141	  if (msg.role === 'assistant' && msg.tool_calls && msg.tool_calls.length > 0) {
   142	    return {
   143	      role: 'assistant',
   144	      content: msg.content,
   145	      tool_calls: msg.tool_calls.map((tc) => ({
   146	        id: tc.id,
   147	        type: 'function',
   148	        function: {
   149	          name: toWireToolName(tc.name),
   150	          arguments: JSON.stringify(tc.arguments),
   151	        },
   152	      })),
   153	    };
   154	  }
   155	  return { role: msg.role, content: msg.content };
   156	}
   157	
   158	function toOpenAITool(
   159	  tool: LlmToolDef,
   160	  _wireMap: ReadonlyMap<string, string>,
   161	): {
   162	  type: 'function';
   163	  function: { name: string; description: string; parameters: Record<string, unknown> };
   164	} {
   165	  return {
   166	    type: 'function',
   167	    function: {
   168	      name: toWireToolName(tool.name),
   169	      description: tool.description,
   170	      parameters: tool.parameters,
   171	    },
   172	  };
   173	}
   174	
   175	function parseToolCall(
   176	  raw: OpenAIToolCall,
   177	  wireMap: ReadonlyMap<string, string>,
   178	): { id: string; name: string; arguments: Record<string, unknown> } {
   179	  let parsedArgs: Record<string, unknown> = {};
   180	  const rawArgs = raw.function.arguments;
   181	  try {
   182	    parsedArgs =
   183	      typeof rawArgs === 'string' ? JSON.parse(rawArgs) : (rawArgs as Record<string, unknown>);
   184	  } catch {
   185	    parsedArgs = {};
   186	  }
   187	  return {
   188	    id: raw.id,
   189	    name: fromWireToolName(raw.function.name, wireMap),
   190	    arguments: parsedArgs,
   191	  };
   192	}

   194	function errorMessage(text: string): LlmMessage {
   195	  return { role: 'assistant', content: `[runtime llm error] ${text}` };
   196	}
